    <script>
        const tg = window.Telegram.WebApp;
        tg.expand();

        if (tg.initDataUnsafe?.user) {
            document.getElementById('username').innerText = tg.initDataUnsafe.user.first_name;
        }

        // Конфігурація рівнів відеокарт
        const GPU_TIERS = {
            1: { name: 'GTX 1060', hash: 2, icon: '📟' },
            2: { name: 'RTX 3060', hash: 5, icon: '📼' },
            3: { name: 'RTX 4090', hash: 14, icon: '🎛' },
            4: { name: 'ASIC Miner', hash: 40, icon: '🖥' },
            5: { name: 'Quantum Rig', hash: 120, icon: '⚛️' }
        };

        // Завантаження стану гри
        let state = JSON.parse(localStorage.getItem('gram_miner_save')) || {
            balance: 50,
            gpus: [], // Масив рівнів карт, напр. [1, 1, 2]
            gpuPrice: 50,
            temp: 30
        };

        const MAX_SLOTS = 6;
        let isOverheated = false;

        // Розрахунок загального хешрейту
        function getTotalHashrate() {
            if (isOverheated) return 0;
            return state.gpus.reduce((sum, lvl) => sum + (GPU_TIERS[lvl]?.hash || 0), 0);
        }

        // Покупка нової карти 1-го рівня
        function buyGPU() {
            if (state.balance >= state.gpuPrice && state.gpus.length < MAX_SLOTS) {
                state.balance -= state.gpuPrice;
                state.gpus.push(1);
                state.gpuPrice = Math.floor(state.gpuPrice * 1.35); // Зростання ціни
                tg.HapticFeedback.notificationOccurred('success');
                saveAndRender();
            }
        }

        // Перевірка наявності пар для крафту
        function findMergeableLevel() {
            const counts = {};
            for (let lvl of state.gpus) {
                if (lvl < 5) { // Максимум 5 рівень
                    counts[lvl] = (counts[lvl] || 0) + 1;
                    if (counts[lvl] >= 2) return lvl;
                }
            }
            return null;
        }

        // Об'єднання двох однакових карт у карту вищого рівня
        function mergeGPUs() {
            const lvlToMerge = findMergeableLevel();
            if (!lvlToMerge) return;

            // Видаляємо 2 карти цього рівня
            let removed = 0;
            state.gpus = state.gpus.filter(lvl => {
                if (lvl === lvlToMerge && removed < 2) {
                    removed++;
                    return false;
                }
                return true;
            });

            // Додаємо 1 карту наступного рівня
            state.gpus.push(lvlToMerge + 1);
            state.gpus.sort((a, b) => b - a); // Сортуємо від потужніших до слабших
            tg.HapticFeedback.impactOccurred('heavy');
            saveAndRender();
        }

        // Охолодження ферми
        function coolRig() {
            state.temp = Math.max(25, state.temp - 15);
            if (state.temp < 80) isOverheated = false;
            tg.HapticFeedback.impactOccurred('light');
            updateUI();
        }

        // Відображення сітки слотів
        function renderGrid() {
            const grid = document.getElementById('rigGrid');
            grid.innerHTML = '';

            for (let i = 0; i < MAX_SLOTS; i++) {
                const slot = document.createElement('div');
                const lvl = state.gpus[i];

                if (lvl) {
                    const info = GPU_TIERS[lvl];
                    slot.className = 'slot filled';
                    slot.innerHTML = `
                        <div class="gpu-icon">${info.icon}</div>
                        <div class="gpu-lvl">Lvl ${lvl} • ${info.name}</div>
                        <div class="gpu-hash">+${info.hash} H/s</div>
                    `;
                } else {
                    slot.className = 'slot';
                    slot.innerHTML = `<span style="color: #484f58; font-size: 12px;">Порожній слот</span>`;
                }
                grid.appendChild(slot);
